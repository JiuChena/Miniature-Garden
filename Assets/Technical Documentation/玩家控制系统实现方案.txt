  public class TeamManager : MonoBehaviour
  {
      [SerializeField] private UnitAssetInformation[] _assetInfos; // 4涓猄O閰嶇疆

      public CharacterAnimatorDriver[] Drivers { get; private set; } = new CharacterAnimatorDriver[4];
      public int CurrentIndex { get; private set; } = 0;
      public CharacterAnimatorDriver CurrentDriver => Drivers[CurrentIndex];

      private float _switchCooldown;
      private const float SwitchCooldownTime = 0.5f;

      // 娓告垙寮€濮嬫椂寮傛鍔犺浇鍏ㄩ儴瑙掕壊
      public async Task LoadTeamAsync()
      {
          for (int i = 0; i < _assetInfos.Length; i++)
          {
              if (_assetInfos[i] == null) continue;
              GameObject obj = await AddressableManager.Instance.LoadAssetAsync<GameObject>
                                  (_assetInfos[i].assetID);
              GameObject instance = Instantiate(obj, spawnPoints[i].position, Quaternion.identity);
              Drivers[i] = instance.GetComponent<CharacterAnimatorDriver>();
              Drivers[i].Initialize(_assetInfos[i]);
              // 闈炲綋鍓嶈鑹插垵濮嬬鐢ㄨ緭鍏?
              if (i != 0) Drivers[i].SetInputEnabled(false);
          }
      }

      private void Update()
      {
          if (_switchCooldown > 0) _switchCooldown -= Time.deltaTime;
      }

      public void TrySwitchTo(int index)
      {
          if (index == CurrentIndex) return;
          if (index < 0 || index >= Drivers.Length) return;
          if (Drivers[index] == null) return;
          if (Drivers[index].ActualData.IsDead) return;
          if (_switchCooldown > 0) return;

          Drivers[CurrentIndex].SetInputEnabled(false);
          CurrentIndex = index;
          Drivers[CurrentIndex].SetInputEnabled(true);
          _switchCooldown = SwitchCooldownTime;

          EventCenter.Instance.SetEventTrigger(EventNames.CharacterSwitched, CurrentIndex);
      }

      // 褰撳墠瑙掕壊姝讳骸鏃惰嚜鍔ㄥ垏鎹?
      public void OnCurrentCharacterDeath()
      {
          for (int i = 0; i < Drivers.Length; i++)
          {
              if (i != CurrentIndex && Drivers[i] != null && !Drivers[i].ActualData.IsDead)
              {
                  TrySwitchTo(i);
                  return;
              }
          }
          // 鍏ㄧ伃
          EventCenter.Instance.SetEventTrigger(EventNames.GameOver);
      }
  }

  public class EnergySystem : MonoBehaviour
  {
      public const float MaxEnergy = 10f;
      private float _curEnergy;
      private float _efficiency = 1f;   // 鍙 Buff 淇敼

      public float CurrentEnergy => _curEnergy;
      public bool IsFull => _curEnergy >= MaxEnergy;

      // 閫犳垚浼ゅ鏃剁Н绱兘閲忥紙鐢?GeneralDataHandler/BulletDriverModule 璋冪敤锛?
      public void GainEnergy(float amount)
      {
          _curEnergy = Mathf.Min(MaxEnergy, _curEnergy + amount * _efficiency);
          EventCenter.Instance.SetEventTrigger(EventNames.EnergyChanged, _curEnergy);
      }

      // 鐖嗗彂鎶€鑳借Е鍙戝墠妫€鏌ュ苟娑堣€?
      public bool TryConsume(float cost)
      {
          if (_curEnergy < cost) return false;
          _curEnergy -= cost;
          EventCenter.Instance.SetEventTrigger(EventNames.EnergyChanged, _curEnergy);
          return true;
      }

      public void SetEfficiency(float value) => _efficiency = value;
  }

  public class PlayerControlModule : MonoBehaviour
  {
      public static PlayerControlModule Instance { get; private set; }

      public PlayerInputManager  InputManager      { get; private set; }
      public TeamManager         TeamManager       { get; private set; }
      public PlayerCameraController CameraController { get; private set; }
      public EnergySystem        EnergySystem      { get; private set; }

      private void Awake()
      {
          Instance = this;
          InputManager      = GetComponent<PlayerInputManager>();
          TeamManager       = GetComponent<TeamManager>();
          CameraController  = GetComponent<PlayerCameraController>();
          EnergySystem      = GetComponent<EnergySystem>();
      }

      private async void Start()
      {
          await TeamManager.LoadTeamAsync();
          CameraController.SwitchToCharacter(TeamManager.CurrentDriver);
          EventCenter.Instance.AddEventListener<int>(EventNames.CharacterSwitched, OnCharacterSwitched);
          EventCenter.Instance.AddEventListener<GameObject>(EventNames.UnitDeath, OnUnitDeath);
      }

      private void Update()
      {
          var data = InputManager.Data;

          // 瑙掕壊鍒囨崲
          if (data.SwitchIndex >= 0)
              TeamManager.TrySwitchTo(data.SwitchIndex);
          else if (data.ScrollDelta != 0)
              TeamManager.TrySwitchTo((TeamManager.CurrentIndex - data.ScrollDelta + 4) % 4);
      }

      private void OnCharacterSwitched(int index)
          => CameraController.SwitchToCharacter(TeamManager.Drivers[index]);

      private void OnUnitDeath(GameObject unit)
      {
          if (unit == TeamManager.CurrentDriver.gameObject)
              TeamManager.OnCurrentCharacterDeath();
      }

      // 瀵瑰蹇嵎璁块棶
      public CharacterAnimatorDriver CurrentDriver => TeamManager.CurrentDriver;
      public PlayerInputData InputData => InputManager.Data;

      private void OnDestroy()
      {
          EventCenter.Instance.RemoveEventListener<int>(EventNames.CharacterSwitched, OnCharacterSwitched);
          EventCenter.Instance.RemoveEventListener<GameObject>(EventNames.UnitDeath, OnUnitDeath);
      }
  }
