 public class BinaryDataManager
 {
     private static readonly BinaryDataManager _instance = new();
     public static BinaryDataManager Instance => _instance;

     private static readonly string DataPath =
         Application.persistentDataPath + "/Data/";

     public void Save<T>(string path, string fileName, T data)
     {
         string fullPath = DataPath + path;
         if (!Directory.Exists(fullPath)) Directory.CreateDirectory(fullPath);
         byte[] bytes = MessagePackSerializer.Serialize(data);
         File.WriteAllBytes(fullPath + fileName + ".bin", bytes);
     }

     public T Load<T>(string path, string fileName)
     {
         string fullPath = DataPath + path + fileName + ".bin";
         if (!File.Exists(fullPath)) return default;
         byte[] bytes = File.ReadAllBytes(fullPath);
         return MessagePackSerializer.Deserialize<T>(bytes);
     }
 }

 public class HSM
 {
     public IStateOwner Owner { get; }
     public HSM(IStateOwner owner) { Owner = owner; }

     private readonly Dictionary<Type, StateBase> _states = new();
     private StateBase _currentState;

     public void AddState<T>(StateBase state) where T : StateBase
         => _states.TryAdd(typeof(T), state);

     public void SwitchState<T>() where T : StateBase
     {
         if (!_states.TryGetValue(typeof(T), out var next)) return;
         if (_currentState == next) return;
         _currentState?.OnExit();
         _currentState = next;
         _currentState.OnEnter();
     }

     public void Tick() => _currentState?.OnUpdate();
     public StateBase CurrentState => _currentState;
 }
